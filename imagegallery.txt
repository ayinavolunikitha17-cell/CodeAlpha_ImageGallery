<script>

let images=[];
let currentFilter="All";
let searchText="";

/* FIXED IMAGE DATA (NO BROKEN LINKS) */
function generate(){
  let data=[
    {cat:"Nature",src:"https://images.unsplash.com/photo-1501785888041-af3ef285b470"},
    {cat:"Nature",src:"https://images.unsplash.com/photo-1441974231531-c6227db76b6e"},
    {cat:"Animals",src:"https://images.unsplash.com/photo-1507149833265-60c372daea22"},
    {cat:"Animals",src:"https://images.unsplash.com/photo-1552053831-71594a27632d"},
    {cat:"Travel",src:"https://images.unsplash.com/photo-1507525428034-b723cf961d3e"},
    {cat:"Travel",src:"https://images.unsplash.com/photo-1469474968028-56623f02e42e"},
    {cat:"Food",src:"https://images.unsplash.com/photo-1504674900247-0877df9cc836"},
    {cat:"Food",src:"https://images.unsplash.com/photo-1546069901-ba9599a7e63c"},
    {cat:"Cars",src:"https://images.unsplash.com/photo-1492144534655-ae79c964c9d7"},
    {cat:"Cars",src:"https://images.unsplash.com/photo-1503376780353-7e6692767b70"},
    {cat:"City",src:"https://images.unsplash.com/photo-1491884662610-dfcd28f30cfb"},
    {cat:"City",src:"https://images.unsplash.com/photo-1480714378408-67cf0d13bc1b"},
    {cat:"Tech",src:"https://images.unsplash.com/photo-1518770660439-4636190af475"},
    {cat:"Tech",src:"https://images.unsplash.com/photo-1550751827-4bd374c3f58b"}
  ];

  data.forEach(i=>{
    images.push({
      src:i.src+"?auto=format&fit=crop&w=600&q=80",
      cat:i.cat,
      fav:false
    });
  });
}

/* RENDER */
function render(){
  let gallery=document.getElementById("gallery");
  gallery.innerHTML="";

  let filtered=images.filter(img=>{
    return (currentFilter==="All" || img.cat===currentFilter)
    && img.cat.toLowerCase().includes(searchText);
  });

  filtered.forEach((img,i)=>{
    let div=document.createElement("div");
    div.className="card";

    div.innerHTML=`
      <img src="${img.src}" loading="lazy">
      <div class="badge">${img.cat}</div>
      <div class="fav">${img.fav?"❤️":"🤍"}</div>
    `;

    div.querySelector(".fav").onclick=(e)=>{
      e.stopPropagation();
      img.fav=!img.fav;
      render();
    };

    gallery.appendChild(div);
  });
}

/* FILTER */
document.querySelectorAll(".filters button").forEach(btn=>{
  btn.onclick=()=>{
    document.querySelectorAll(".filters button")
    .forEach(b=>b.classList.remove("active"));

    btn.classList.add("active");
    currentFilter=btn.getAttribute("data");
    render();
  };
});

/* SEARCH */
document.getElementById("search").oninput=(e)=>{
  searchText=e.target.value.toLowerCase();
  render();
};

/* LOAD MORE */
function loadMore(){
  generate();
  render();
}

/* INIT */
generate();
generate();
render();

</script>
